1. Get the Forgejo URL:

{{- if $ingressEnabled }}
  {{- if $ingressTLS }}
  https://{{ $ingressHost }}/
  {{- else }}
  http://{{ $ingressHost }}/
  {{- end }}

  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        Watch the status with: 'kubectl get ingress --namespace {{ .Release.Namespace }} -w {{ $fullName }}'

{{- else if contains "NodePort" .Values.service.type }}

  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ $fullName }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "http://${NODE_IP}:${NODE_PORT}"

{{- else if contains "LoadBalancer" .Values.service.type }}

  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        Watch the status with: 'kubectl get svc --namespace {{ .Release.Namespace }} -w {{ $fullName }}'

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ $fullName }} --template "{{ "{{ range (index .status.loadBalancer.ingress 0) }}{{ . }}{{ end }}" }}")
  echo "http://${SERVICE_IP}:{{ $httpPort }}"

{{- else if contains "ClusterIP" .Values.service.type }}

  kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ $fullName }} {{ $httpPort }}:{{ $httpPort }} &
  echo "http://127.0.0.1:{{ $httpPort }}"

{{- end }}

    --env="PGPASSWORD=$(kubectl get secret --namespace {{ .Release.Namespace }} {{ $fullName }}-postgresql -o jsonpath="{.data.password}" | base64 -d)" \
    --command -- psql --host {{ $fullName }}-postgresql-rw -U {{ .Values.postgresql.database.username }} -d {{ .Values.postgresql.database.name }} -p 5432
